    55→    isGenerating, setIsGenerating,

    57→    settings, activeTab, setActiveTab,
    58→  } = useSiscuniaStore()
    59→
    60→  const [sendingToNotion, setSendingToNotion] = useState<string | null>(null)
    61→  const [editingVersion, setEditingVersion] = useState<number | null>(null)
    62→  const [editedText, setEditedText] = useState('')

    70→    if (!settings?.googleApiKeySet) {
    71→      toast.error('Configura tu API Key de Google AI Studio primero')

    73→      return

    75→    setIsGenerating(true)
    76→    setGeneratedResult(null)
    77→    try {
    78→      const res = await fetch('/api/generate', {
    79→        method: 'POST',
    80→        headers: { 'Content-Type': 'application/json' },
    81→        body: JSON.stringify(formData),
    82→      })
    83→      const data = await res.json()
    84→      if (data.success) {
    85→        setGeneratedResult(data.script)
    86→        toast.success('Libreto generado exitosamente')
    87→      } else {
    88→        toast.error(data.error || 'Error al generar')
    89→      }
    90→    } catch {
    91→      toast.error('Error de conexión con el servidor')
    92→    } finally {
    93→      setIsGenerating(false)
    94→    }
    95→  }
    96→
    97→  const handleSendToNotion = async (versionIndex: number) => {
    98→    if (!generatedResult?.id) return

   100→      toast.error('Configura el Token de Notion y el ID de la base de datos primero')
   101→      setActiveTab('config')

   103→    }
   104→    setSendingToNotion(String(versionIndex))

   106→      const contentToSend = editingVersion === versionIndex ? editedText : undefined
   107→      const res = await fetch('/api/send-to-notion', {
   108→        method: 'POST',
   109→        headers: { 'Content-Type': 'application/json' },
   110→        body: JSON.stringify({ scriptId: generatedResult.id, versionIndex, editedContent: contentToSend }),
   111→      })
   112→      const data = await res.json()
   113→      if (data.success) {
   114→        toast.success('Enviado a Notion correctamente')
   115→        setLastNotionUrl(data.notionUrl)
   116→      } else {
   117→        toast.error(data.error || 'Error al enviar a Notion')
   118→      }
   119→    } catch {
   120→      toast.error('Error de conexión con Notion')
   121→    } finally {
   122→      setSendingToNotion(null)
   123→    }
   124→  }
   125→
   126→  const handleCopy = (text: string) => {
   127→    navigator.clipboard.writeText(text)
   128→    toast.success('Copiado al portapapeles')
   129→  }
   130→
   131→  const startEditing = (idx: number, text: string) => {
   132→    setEditingVersion(idx)
   133→    setEditedText(text)
   134→  }
   135→
   136→  const saveEditing = () => {
   137→    if (generatedResult?.versiones && editingVersion !== null) {
   138→      const updated = { ...generatedResult }
   139→      updated.versiones = [...updated.versiones]
   140→      updated.versiones[editingVersion] = {
   141→        ...updated.versiones[editingVersion],
   142→        libreto: editedText,
   143→      }
   144→      setGeneratedResult(updated)
   145→    }
   146→    setEditingVersion(null)
   147→    toast.success('Versión actualizada')

   172→            </div>
   173→          </div>
   174→        </CardHeader>
